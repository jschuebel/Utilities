ALTER PROCEDURE [base].[CheckDeltas]

As

BEGIN

 

       --Kept for test dump only output 0=do no update/inserts

       DECLARE @runUpdates bit=1

       DECLARE @LatestSysStartTime dateTime2(7)

       --Add 1 millisecond to avoid the previous latest record to be processed again

       SELECT @LatestSysStartTime=DateAdd(ms,1,[DateValue]) FROM [base].[JobSetting] bs WHERE CODE='MaxStart' AND [IsActive]=1

 

       declare @toDate varchar(38),@frmDate varchar(38) = CONVERT(VARCHAR(38), @LatestSysStartTime, 121)

       --Gather date w/o time to append end of day time

       set @toDate = TRIM(substring(@frmDate,0,charindex(' ', @frmDate)))

       set @toDate = @toDate + ' 23:59:59.9999999'

       if @runUpdates=0

              select @frmDate fromDate, @toDate toDate

 

 

       declare @Parent TABLE (

              [DependentKeyID] [int] NOT NULL,

              [FirstName] [varchar](50) NOT NULL,

              [LastName] [varchar](50) NOT NULL,

              [ExistRecs] int NULL,

              [Deleted] bit NULL

              )

       ;

      

Create CTE

 

 

       insert into @Parent

       select  *, [ExistRecs],[Deleted] from (

              select *

                     ,ROW_NUMBER() OVER (PARTITION BY ID ORDER BY SysStartTime DESC) AS rev

                     --Flag records that exist

                     ,(select count(*) FROM) ExistRecs

                     --Flag records that exist in CurrentTable, but not in ParentTable

                     ,(CASE WHEN (select count(*) FROM [Parent] per (nolock) WHERE per.ID=cte.ID)=0 THEN 1 ELSE 0 END) Deleted

              from CTE cte

       ) as xDat WHERE xDat.rev=1

 

       if @runUpdates=0

              select * from @Parent

 

       if @runUpdates=1

       BEGIN

              --Updated Rows  -- Ranked because you can have multiple during the time period, so take the latest (rev=1)

              UPDATE per

              SET per.

              FROM [LookUp] per

              inner join @Parent pp on pp.ID=per.ID AND [ExistRecs]>0 AND [Deleted]=0

 

              --Insert New Rows

              INSERT INTO [LookUp] ()

              select

              FROM @Parent WHERE [ExistRecs]=0 AND [Deleted]=0

 

              --Remove Deleted Rows

              DELETE FROM LookUp WHERE ID in (

                     SELECT LookUp

                     WHERE [ID] NOT IN

                           (SELECT [ID] FROM Parent (nolock)))

       END

       --Update the setting for the next LookUp update

       --Never go backwards, i.e. latest record was removed. (it would show in history still)

       DECLARE @LookUpLatest dateTime2(7), @ParentLatest dateTime2(7)

       SELECT @LookUpLatest=[DateValue] FROM [base].[JobSetting] bs WHERE CODE='MaxStart' AND [IsActive]=1

       SELECT @ParentLatest=MAX(SysStartTime) FROM [Parent]

       if (@LookUpLatest < @ParentLatest)

       BEGIN

              if @runUpdates=1

              BEGIN

                     UPDATE bs

                           SET [DateValue]=(select MAX(SysStartTime) FROM [Parent])

                     FROM [base].[JobSetting] bs

                     WHERE CODE='MaxStart' AND [IsActive]=1

              END

       END

 

END
